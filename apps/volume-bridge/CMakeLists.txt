cmake_minimum_required(VERSION 3.20)
project(VolumeBridge LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/libaspl")
file(GLOB ASPL_SOURCES CONFIGURE_DEPENDS "${VENDOR}/src/*.cpp")
add_library(aspl STATIC ${ASPL_SOURCES})
target_include_directories(aspl PUBLIC "${VENDOR}/include")
target_compile_options(aspl PRIVATE -Wno-invalid-offsetof -Wno-deprecated-declarations)
target_link_libraries(aspl PUBLIC "-framework CoreFoundation")
add_library(VolumeBridge MODULE driver/Driver.cpp)
target_include_directories(VolumeBridge PRIVATE native)
target_compile_options(VolumeBridge PRIVATE -fblocks -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(VolumeBridge PRIVATE aspl "-framework CoreFoundation")
set_target_properties(VolumeBridge PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION driver PREFIX "" SUFFIX ""
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Driver-Info.plist.in")
add_library(bridge_native STATIC native/Service.mm native/MonitorVolume.mm)
target_include_directories(bridge_native PUBLIC native)
target_compile_options(bridge_native PRIVATE -fblocks -Wall -Wextra -Wno-unused-parameter)
add_executable(ring-test tests/native/ring.cpp)
target_include_directories(ring-test PRIVATE native)
target_compile_options(ring-test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME ring COMMAND ring-test)
add_executable(volume-test tests/native/volume.cpp)
target_include_directories(volume-test PRIVATE driver)
target_link_libraries(volume-test PRIVATE aspl)
target_compile_options(volume-test PRIVATE -UNDEBUG)
add_test(NAME volume COMMAND volume-test)
